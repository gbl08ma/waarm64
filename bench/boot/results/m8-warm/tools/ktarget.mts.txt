// Kernel entry-stall samples by what the target block is: a return continuation (after BL/BLR), a symbol's start, other.
import { readFileSync } from 'node:fs';
const [img, map, jit] = process.argv.slice(2);
const image = readFileSync(img);
const TEXT = 0xffff800080000000n;
const starts = new Set(readFileSync(map, 'utf8').split('\n').map((l) => l.split(' ')).filter((r) => r.length === 3 && /[tTwW]/.test(r[1])).map((r) => r[0]));
const n = new Map<string, number>();
let tot = 0;
for (const l of readFileSync(jit, 'utf8').split('\n')) {
  const r = l.split('\t');
  const m = /^JS:0x([0-9a-f]{16})_pa/.exec(r[1] ?? '');
  if (!m || r[2] !== '0') continue;
  tot++;
  const va = BigInt('0x' + m[1]);
  const off = Number(va - TEXT);
  const prev = off >= 4 && off < image.length ? image.readUInt32LE(off - 4) : 0;
  const kind = off < 4 || off >= image.length ? 'outside the image (modules, BPF)' : ((prev & 0xfc000000) >>> 0) === 0x94000000 || ((prev & 0xfffffc1f) >>> 0) === 0xd63f0000 ? 'return continuation (after BL/BLR)' : starts.has(m[1]) || image.readUInt32LE(off) === 0xd503233f || image.readUInt32LE(off) === 0xd503245f ? 'a function start' : 'other block';
  const how = r[8] === '8000003' && r[9] === 'x17=pc' ? 'indirect tail' : r[8] === '8000003' ? 'dispatcher' : 'other';
  n.set(`${kind}, ${how}`, (n.get(`${kind}, ${how}`) ?? 0) + 1);
}
console.log(`kernel entry stalls: ${tot}`);
for (const [k, v] of [...n].sort((a, b) => b[1] - a[1])) console.log(`${((100 * v) / tot).toFixed(1).padStart(5)}%  ${k}`);
