// Kernel guest-code samples by the kernel symbol of the region's head block (System.map).
import { readFileSync } from 'node:fs';
const [map, jit] = process.argv.slice(2);
const syms = readFileSync(map, 'utf8').split('\n').map((l) => l.split(' ')).filter((r) => r.length === 3 && /[tTwW]/.test(r[1])).map((r) => [BigInt('0x' + r[0]), r[2]] as const).sort((a, b) => (a[0] < b[0] ? -1 : 1));
const find = (va: bigint) => { let lo = 0, hi = syms.length; while (lo < hi) { const m = (lo + hi) >> 1; if (syms[m][0] <= va) lo = m + 1; else hi = m; } return lo ? syms[lo - 1][1] : '?'; };
const n = new Map<string, [number, number]>();
let tot = 0, all = 0;
for (const l of readFileSync(jit, 'utf8').split('\n')) {
  const r = l.split('\t');
  const m = /^JS:0x([0-9a-f]{16})_pa/.exec(r[1] ?? '');
  if (/_pa0x/.test(r[1] ?? '')) all++;
  if (!m) continue;
  tot++;
  const s = find(BigInt('0x' + m[1]));
  const v = n.get(s) ?? n.set(s, [0, 0]).get(s)!;
  v[0]++; if (r[2] === '0') v[1]++;
}
console.log(`kernel ${tot} of ${all} guest-code samples`);
for (const [s, [a, e]] of [...n].sort((x, y) => y[1][0] - x[1][0]).slice(0, 45)) console.log(`${((100 * a) / tot).toFixed(2).padStart(6)}% ${((100 * e) / a).toFixed(0).padStart(3)}% entry  ${s}`);
