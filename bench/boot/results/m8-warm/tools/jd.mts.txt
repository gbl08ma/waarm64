// Resolves perf samples (perf script -F tid,time,ip,dso) against V8's jitdump (--perf-prof) with time, so code
// space reused after a GC resolves to the code that was there at the sample's time. Writes, per sample in JIT code:
// tid, function name, offset, the word at the IP and the 3 words before it.
//   node jd.ts <jit-*.dump dir> <samples.txt> <out.tsv>
import { fstatSync, openSync, readFileSync, readSync, readdirSync, writeFileSync } from 'node:fs';
const [dir, samplesFile, outFile] = process.argv.slice(2);
type Rec = { addr: number; size: number; name: string; ts: number; fd: number; off: number };
const recs: Rec[] = [];
for (const f of readdirSync(dir).filter((f) => f.startsWith('jit-') && f.endsWith('.dump'))) {
  const fd = openSync(`${dir}/${f}`, 'r');
  const fileLen = fstatSync(fd).size;
  const W = 64 << 20;
  const buf = Buffer.alloc(W);
  let base = 0, have = readSync(fd, buf, 0, W, 0);
  if (buf.readUInt32LE(0) !== 0x4a695444) throw new Error(`${f}: bad magic`);
  let p = buf.readUInt32LE(8);
  // Only the header of each record is needed (name up to 4 KiB): keep the window ahead of p by 8 KiB.
  while (base + p + 16 <= fileLen) {
    if (p + 8192 > have && base + have < fileLen) {
      base += p; have = readSync(fd, buf, 0, W, base); p = 0;
    }
    const id = buf.readUInt32LE(p), total = buf.readUInt32LE(p + 4), ts = Number(buf.readBigUInt64LE(p + 8));
    if (total === 0 || base + p + total > fileLen) break;
    if (id === 0) {
      const addr = Number(buf.readBigUInt64LE(p + 24)), size = Number(buf.readBigUInt64LE(p + 40));
      const nameStart = p + 56, nameEnd = buf.indexOf(0, nameStart);
      recs.push({ addr, size, name: buf.toString('latin1', nameStart, nameEnd), ts, fd, off: base + nameEnd + 1 });
    }
    p += total;
  }
  console.error(`${f}: ${recs.length} code records so far`);
}
recs.sort((a, b) => a.ts - b.ts);
const samples: { tid: number; t: number; ip: number }[] = [];
for (const l of readFileSync(samplesFile, 'utf8').split('\n')) {
  const m = /^\s*(\d+)\s+(\d+)\.(\d+):\s+([0-9a-f]+)/.exec(l);
  if (m) samples.push({ tid: +m[1], t: Number(m[2]) * 1e9 + Number(m[3].padEnd(9, '0')), ip: parseInt(m[4], 16) });
}
samples.sort((a, b) => a.t - b.t);
// Pages of 4 KiB -> the records covering them, latest last.
const pages = new Map<number, Rec[]>();
let ri = 0;
const out: string[] = [];
let hit = 0;
const hot = new Map<Rec, Map<number, number>>();
for (const s of samples) {
  while (ri < recs.length && recs[ri].ts <= s.t) {
    const r = recs[ri++];
    if (r.size > 1 << 22) { console.error(`skip ${r.name} ${r.size}`); continue; }
    for (let pg = Math.floor(r.addr / 4096); pg <= Math.floor((r.addr + r.size - 1) / 4096); pg++) {
      const l = (pages.get(pg) ?? pages.set(pg, []).get(pg)!).filter((o) => o.addr + o.size <= r.addr || o.addr >= r.addr + r.size);
      l.push(r);
      pages.set(pg, l);
    }
  }
  const l = pages.get(Math.floor(s.ip / 4096));
  const r = l?.findLast((o) => s.ip >= o.addr && s.ip < o.addr + o.size);
  if (!r) continue;
  hit++;
  const hk = hot.get(r) ?? hot.set(r, new Map()).get(r)!;
  hk.set(s.ip - r.addr, (hk.get(s.ip - r.addr) ?? 0) + 1);
  const o = s.ip - r.addr;
  const wb = Buffer.alloc(16);
  const start = Math.max(0, o - 12);
  readSync(r.fd, wb, 0, Math.min(16, r.size - start), r.off + start);
  const w = (k: number) => (o - 4 * k >= 0 && o - 4 * k + 4 <= r.size ? wb.readUInt32LE(o - 4 * k - start).toString(16).padStart(8, '0') : '-');
  out.push(`${s.tid}\t${r.name}\t${o}\t${r.size}\t${w(0)}\t${w(1)}\t${w(2)}\t${w(3)}`);
}
console.error(`${samples.length} samples, ${hit} in JIT code`);
writeFileSync(outFile, out.join('\n') + '\n');
// The hottest 40 guest-code functions: code as hex words with per-offset sample counts (annotate with llvm-mc).
const tops = [...hot].filter(([r]) => /_pa0x/.test(r.name)).map(([r, m]) => [r, m, [...m.values()].reduce((a, b) => a + b, 0)] as const).sort((a, b) => b[2] - a[2]).slice(0, 40);
const lines: string[] = [];
for (const [r, m, n] of tops) {
  const b = Buffer.alloc(r.size);
  readSync(r.fd, b, 0, r.size, r.off);
  lines.push(`## ${r.name} size ${r.size} samples ${n}`);
  for (let o = 0; o + 4 <= r.size; o += 4) lines.push(`${o}\t${m.get(o) ?? 0}\t${b.readUInt32LE(o).toString(16).padStart(8, '0')}`);
}
writeFileSync(outFile + '.hot', lines.join('\n') + '\n');
