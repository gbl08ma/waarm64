import { readFileSync } from 'node:fs';
const [map, jit] = process.argv.slice(2);
const syms = readFileSync(map, 'utf8').split('\n').map((l) => l.split(' ')).filter((r) => r.length === 3 && /[tTwW]/.test(r[1])).map((r) => [BigInt('0x' + r[0]), r[2]] as const).sort((a, b) => (a[0] < b[0] ? -1 : 1));
const find = (va: bigint) => { let lo = 0, hi = syms.length; while (lo < hi) { const m = (lo + hi) >> 1; if (syms[m][0] <= va) lo = m + 1; else hi = m; } return lo ? syms[lo - 1][1] : '?'; };
const B: [string, RegExp][] = [
  ['preempt/smp debug checks', /^(debug_smp_processor_id|check_preemption_disabled|__this_cpu_preempt_check|preempt_count_(add|sub))$/],
  ['psi', /^psi_/],
  ['tick, hrtimer, timekeeping', /tick|hrtimer|timer|timekeeping|clock|cntvct|ktime|jiffies|calc_global_load|account_process|update_process_times|rcu_sched_clock_irq|run_local_timers|update_wall|arch_timer|irqtime|account_|vtime|cpuacct|cgroup_account/],
  ['scheduler', /sched|enqueue|dequeue|load_avg|update_curr|pick_next|put_prev|set_next|wake|reweight|task_rq|select_task_rq|update_sg|balance|newidle|_fair$|_rt$|_dl$|cfs|rq_|^yield|context_switch|finish_task|switch_to|cpu_switch|fpsimd|update_blocked|idle/],
  ['irq entry, gic, softirq', /irq|gic|vectors|el1h|el0t|el0_|el1_|entry|handle_|ret_to_user|exit_to_user|softirq|do_interrupt|ipi/],
  ['rcu', /rcu/],
  ['futex, syscalls, signals', /futex|syscall|invoke|signal|do_sys|__arm64_sys|__se_sys|__do_sys|ksys/],
  ['memory: faults, mm, slab, memcg', /fault|page|folio|mm_|mas_|mte|vma|pte|pmd|slab|kmem|kmalloc|kfree|memcg|alloc|free|lru|rmap|tlb|flush|memset|memcpy|copy|clear|__pi_mem/],
  ['net, unix, ipc', /unix|sock|skb|tcp|udp|ip_|net|inet|sk_/],
  ['fs, poll, pipes, eventfd', /vfs|file|poll|epoll|fd|pipe|inode|dentry|path|read|write|iter|ep_|fsnotify|eventfd|ext4|seq_/],
  ['locks', /lock|mutex|sem|spin/],
  ['stack traces (debug)', /stack|unwind|kunwind/],
];
const n = new Map<string, number>();
let tot = 0;
for (const l of readFileSync(jit, 'utf8').split('\n')) {
  const r = l.split('\t');
  const m = /^JS:0x([0-9a-f]{16})_pa/.exec(r[1] ?? '');
  if (!m) continue;
  tot++;
  const s = find(BigInt('0x' + m[1]));
  const b = B.find(([, re]) => re.test(s))?.[0] ?? 'other';
  n.set(b, (n.get(b) ?? 0) + 1);
}
for (const [k, v] of [...n].sort((a, b) => b[1] - a[1])) console.log(`${((100 * v) / tot).toFixed(1).padStart(5)}%  ${k}`);
