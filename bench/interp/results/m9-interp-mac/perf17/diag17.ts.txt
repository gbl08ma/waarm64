// Prints the last `diag17` line of a results.json (or a log) by class, most exits first.
//   node diag17.ts <results.json|log> [rows]
import { readFileSync } from 'node:fs';
const f = process.argv[2], rows = Number(process.argv[3] ?? 40);
const t = readFileSync(f, 'utf8');
const lines = f.endsWith('.json') ? (JSON.parse(t).slow as string[]).filter((l) => l.includes('diag17 ')) : t.split('\n').filter((l) => l.includes('diag17 '));
const p = lines[lines.length - 1].slice(lines[lines.length - 1].indexOf('diag17 ') + 7).split(' ');
const tot: Record<string, number> = {};
let i = 0;
for (; i + 1 < p.length && !p[i].includes(':'); i += 2) tot[p[i]] = Number(p[i + 1]);
const cls = p.slice(i).map((s) => { const [n, e, u] = s.split(':'); return { n, e: Number(e), u: Number(u) }; }).sort((a, b) => b.e - a.e);
console.log(`exits ${tot.exits} uops ${tot.uops} steps ${tot.steps} (exits per 1000 steps ${(1000 * tot.exits / tot.steps).toFixed(2)}; uops ${(1000 * tot.uops / tot.steps).toFixed(2)})`);
for (const c of cls.slice(0, rows)) console.log(`${c.n.padEnd(22)} ${String(c.e).padStart(11)} ${(100 * c.e / tot.exits).toFixed(1).padStart(5)}%  uops ${String(c.u).padStart(11)} ${(100 * c.u / tot.uops).toFixed(1).padStart(5)}%`);
