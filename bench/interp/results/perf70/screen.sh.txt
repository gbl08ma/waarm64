#!/usr/bin/env bash
# PERF loop 70's interpreter screen: base, sink, pair alternated, the order rotated each round.
cd "$(dirname "$0")/../../../.."
A=bench/results/local/perf70
B=(base sink pair)
for r in 1 2 3; do
  o=("${B[@]:$(( (r-1) % 3 ))}" "${B[@]:0:$(( (r-1) % 3 ))}")
  arms=(); for a in "${o[@]}"; do arms+=("$a=$A/$a.wasm"); done
  echo "round $r: ${o[*]}"
  bench/interp/ab-mac.sh p70s$r 1 "${arms[@]}" -- dec_mips vm_dec_flat_mips vm_dec_mmu_mips boot_alpine_dec
  timeout 60 bench/gpu/mac-leftovers.sh --kill 2>&1 | tail -1
done
echo screen done
